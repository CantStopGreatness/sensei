cmake_minimum_required(VERSION 3.16)
project(Sensei CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Threads REQUIRED)
add_library(core src/bitboard.cpp src/position.cpp src/movegen.cpp src/perft.cpp src/uci.cpp src/eval.cpp src/search.cpp src/tt.cpp)
target_include_directories(core PUBLIC src)
target_link_libraries(core PUBLIC Threads::Threads)

add_executable(sensei src/main.cpp)
target_link_libraries(sensei core)

enable_testing()
add_executable(perft_tests tests/perft_tests.cpp)
target_link_libraries(perft_tests core)
add_test(NAME perft COMMAND perft_tests)
add_executable(search_tests tests/search_tests.cpp)
target_link_libraries(search_tests core)
add_test(NAME search COMMAND search_tests)
